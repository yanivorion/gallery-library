function Component({ config = {} }) {
  const [isOpen, setIsOpen] = React.useState(false);
  const [hoveredBlind, setHoveredBlind] = React.useState(null);
  const [blindStates, setBlindStates] = React.useState([]);
  const [isInView, setIsInView] = React.useState(false);
  const containerRef = React.useRef(null);

  const frontImage = config?.frontImage || MANIFEST.editorElement.data.frontImage.defaultValue;
  const backImage = config?.backImage || MANIFEST.editorElement.data.backImage.defaultValue;
  const blindCount = parseInt(config?.blindCount || '8');
  const containerWidth = parseInt(config?.containerWidth || '700');
  const containerHeight = parseInt(config?.containerHeight || '500');
  const blindGap = parseInt(config?.blindGap || '3');
  const maxRotation = parseInt(config?.maxRotation || '70');
  const waveEffect = config?.waveEffect !== false;
  const waveDelay = parseInt(config?.waveDelay || '50');
  const transitionDuration = parseInt(config?.transitionDuration || '400');
  const backgroundColor = config?.backgroundColor || '#111111';
  const blindEdgeColor = config?.blindEdgeColor || '#444444';
  const lightRayColor = config?.lightRayColor || '#ffffff';
  const perspective = parseInt(config?.perspective || '800');
  const openOnHover = config?.openOnHover !== false;
  const clickToToggle = config?.clickToToggle !== false;
  const showLightRays = config?.showLightRays !== false;

  const prefersReducedMotion = React.useMemo(() => {
    return typeof window !== 'undefined'
      ? window.matchMedia('(prefers-reduced-motion: reduce)').matches
      : false;
  }, []);

  // Initialize blind states
  React.useEffect(() => {
    setBlindStates(Array(blindCount).fill(false));
  }, [blindCount]);

  React.useEffect(() => {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) setIsInView(true);
        });
      },
      { threshold: 0.3 }
    );
    if (containerRef.current) observer.observe(containerRef.current);
    return () => observer.disconnect();
  }, []);

  // Wave effect on hover
  React.useEffect(() => {
    if (!waveEffect || hoveredBlind === null || prefersReducedMotion) return;

    const timeouts = [];
    
    // Propagate wave from hovered blind
    for (let i = 0; i < blindCount; i++) {
      const distance = Math.abs(i - hoveredBlind);
      const delay = distance * waveDelay;
      
      const timeout = setTimeout(() => {
        setBlindStates(prev => {
          const newStates = [...prev];
          newStates[i] = true;
          return newStates;
        });
      }, delay);
      
      timeouts.push(timeout);
    }

    return () => {
      timeouts.forEach(t => clearTimeout(t));
    };
  }, [hoveredBlind, waveEffect, waveDelay, blindCount, prefersReducedMotion]);

  // Reset wave when mouse leaves
  const handleMouseLeave = () => {
    setHoveredBlind(null);
    if (waveEffect && !isOpen) {
      // Reset all blinds with wave effect
      blindStates.forEach((_, i) => {
        setTimeout(() => {
          setBlindStates(prev => {
            const newStates = [...prev];
            newStates[i] = false;
            return newStates;
          });
        }, i * (waveDelay / 2));
      });
    }
  };

  const handleClick = () => {
    if (!clickToToggle) return;
    setIsOpen(!isOpen);
    setBlindStates(Array(blindCount).fill(!isOpen));
  };

  const blindHeight = (containerHeight - (blindCount - 1) * blindGap) / blindCount;

  return (
    <div
      ref={containerRef}
      className="venetian-blinds-gallery"
      style={{
        backgroundColor,
        padding: '60px 40px',
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        fontFamily: "'Neue Haas Grotesk Display Pro', 'Neue Haas Grotesk Text Pro', 'Helvetica Neue', Helvetica, Arial, sans-serif"
      }}
    >
      {/* Blinds Container */}
      <div
        onClick={handleClick}
        onMouseLeave={handleMouseLeave}
        style={{
          position: 'relative',
          width: `${containerWidth}px`,
          height: `${containerHeight}px`,
          perspective: `${perspective}px`,
          cursor: clickToToggle ? 'pointer' : 'default',
          borderRadius: '8px',
          overflow: 'hidden'
        }}
      >
        {/* Back image (revealed when blinds open) */}
        <div
          style={{
            position: 'absolute',
            top: 0,
            left: 0,
            width: '100%',
            height: '100%',
            backgroundImage: `url(${backImage})`,
            backgroundSize: 'cover',
            backgroundPosition: 'center',
            zIndex: 1
          }}
        />

        {/* Light rays when open */}
        {showLightRays && (
          <div
            style={{
              position: 'absolute',
              top: 0,
              left: 0,
              right: 0,
              bottom: 0,
              zIndex: 2,
              pointerEvents: 'none',
              opacity: blindStates.filter(Boolean).length / blindCount * 0.3,
              background: `linear-gradient(180deg, 
                transparent 0%, 
                ${lightRayColor}10 20%, 
                ${lightRayColor}05 40%, 
                transparent 60%,
                ${lightRayColor}08 80%,
                transparent 100%
              )`,
              transition: `opacity ${transitionDuration}ms ease-out`
            }}
          />
        )}

        {/* Blinds */}
        {Array.from({ length: blindCount }).map((_, index) => {
          const isBlindOpen = isOpen || blindStates[index];
          const rotation = isBlindOpen ? maxRotation : 0;
          const yOffset = index * (blindHeight + blindGap);

          return (
            <div
              key={index}
              onMouseEnter={() => openOnHover && setHoveredBlind(index)}
              style={{
                position: 'absolute',
                top: `${yOffset}px`,
                left: 0,
                width: '100%',
                height: `${blindHeight}px`,
                transformStyle: 'preserve-3d',
                transformOrigin: 'center top',
                opacity: prefersReducedMotion ? 1 : (isInView ? 1 : 0),
                transform: prefersReducedMotion
                  ? `rotateX(${rotation}deg)`
                  : (isInView
                    ? `rotateX(${rotation}deg)`
                    : `rotateX(-90deg)`
                  ),
                zIndex: blindCount - index + 10,
                transition: prefersReducedMotion
                  ? 'none'
                  : `transform ${transitionDuration}ms cubic-bezier(0.34, 1.56, 0.64, 1) ${index * 30}ms,
                     opacity ${transitionDuration}ms ease-out ${index * 50}ms`
              }}
            >
              {/* Blind front face */}
              <div
                style={{
                  position: 'absolute',
                  width: '100%',
                  height: '100%',
                  backfaceVisibility: 'hidden',
                  overflow: 'hidden',
                  boxShadow: isBlindOpen
                    ? `0 ${4 + index}px ${8 + index * 2}px rgba(0, 0, 0, 0.3)`
                    : `0 2px 4px rgba(0, 0, 0, 0.2)`,
                  borderBottom: `1px solid ${blindEdgeColor}`
                }}
              >
                {/* Front image slice */}
                <div
                  style={{
                    width: `${containerWidth}px`,
                    height: `${containerHeight}px`,
                    backgroundImage: `url(${frontImage})`,
                    backgroundSize: 'cover',
                    backgroundPosition: `center ${-yOffset}px`,
                    transform: `translateY(${-yOffset}px)`
                  }}
                />

                {/* Blind texture overlay */}
                <div
                  style={{
                    position: 'absolute',
                    top: 0,
                    left: 0,
                    right: 0,
                    bottom: 0,
                    background: `linear-gradient(180deg, 
                      rgba(255, 255, 255, 0.05) 0%, 
                      transparent 30%, 
                      transparent 70%, 
                      rgba(0, 0, 0, 0.1) 100%
                    )`,
                    pointerEvents: 'none'
                  }}
                />
              </div>

              {/* Blind back face (visible when rotated) */}
              <div
                style={{
                  position: 'absolute',
                  width: '100%',
                  height: '100%',
                  backfaceVisibility: 'hidden',
                  transform: 'rotateX(180deg)',
                  backgroundColor: blindEdgeColor,
                  boxShadow: 'inset 0 2px 4px rgba(0, 0, 0, 0.3)'
                }}
              />

              {/* Blind edge highlight */}
              <div
                style={{
                  position: 'absolute',
                  bottom: 0,
                  left: 0,
                  right: 0,
                  height: '2px',
                  background: `linear-gradient(90deg, 
                    transparent 0%, 
                    rgba(255,255,255,${isBlindOpen ? 0.2 : 0.05}) 50%, 
                    transparent 100%
                  )`,
                  pointerEvents: 'none',
                  transition: `opacity ${transitionDuration}ms ease-out`
                }}
              />
            </div>
          );
        })}

        {/* Frame shadow */}
        <div
          style={{
            position: 'absolute',
            top: 0,
            left: 0,
            right: 0,
            bottom: 0,
            boxShadow: 'inset 0 0 30px rgba(0, 0, 0, 0.3)',
            borderRadius: '8px',
            pointerEvents: 'none',
            zIndex: 100
          }}
        />
      </div>

      {/* Controls hint */}
      {(clickToToggle || openOnHover) && (
        <div
          style={{
            position: 'absolute',
            bottom: '20px',
            left: '50%',
            transform: 'translateX(-50%)',
            fontSize: '12px',
            color: '#777777',
            opacity: 0.7
          }}
        >
          {clickToToggle && openOnHover ? 'Hover or click to open blinds' : 
           clickToToggle ? 'Click to toggle blinds' : 
           'Hover to open blinds'}
        </div>
      )}
    </div>
  );
}
