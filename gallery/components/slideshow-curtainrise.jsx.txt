function Component({ config = {} }) {
  const [currentIndex, setCurrentIndex] = React.useState(0);
  const [prevIndex, setPrevIndex] = React.useState(0);
  const [isAnimating, setIsAnimating] = React.useState(false);
  const timerRef = React.useRef(null);

  // Parse Config
  const slides = React.useMemo(() => {
    try {
      return JSON.parse(config?.slidesData || '[]');
    } catch {
      return [];
    }
  }, [config?.slidesData]);

  const autoPlay = config?.autoPlay === true;

  // Animation Logic
  const handleNext = React.useCallback(() => {
    if (isAnimating || slides.length < 2) return;
    setIsAnimating(true);
    setPrevIndex(currentIndex);
    setCurrentIndex((prev) => (prev + 1) % slides.length);
    
    setTimeout(() => {
      setIsAnimating(false);
    }, 1200); // Match transition duration
  }, [currentIndex, isAnimating, slides.length]);

  // Auto Play
  React.useEffect(() => {
    if (autoPlay) {
      timerRef.current = setInterval(handleNext, 4000);
    }
    return () => clearInterval(timerRef.current);
  }, [autoPlay, handleNext]);

  // Handle Empty
  if (slides.length === 0) return <div style={{ padding: 40 }}>Add slides JSON</div>;

  return (
    <div
      onClick={handleNext}
      style={{
        width: '100%',
        height: '100%',
        minHeight: '600px',
        backgroundColor: '#000',
        position: 'relative',
        overflow: 'hidden',
        cursor: 'pointer',
        userSelect: 'none'
      }}
    >
      {/* --- IMAGES LAYER --- */}
      {slides.map((slide, index) => {
        // Logic:
        // Current index is the one sliding UP (entering)
        // Previous index is the one stationary behind, waiting to be covered
        
        let zIndex = 0;
        let transform = 'translateY(100%)'; // Default: waiting below
        let transition = 'none';

        if (index === currentIndex) {
          zIndex = 2; // On top
          transform = 'translateY(0%)'; // Move to cover
          transition = isAnimating ? 'transform 1.2s cubic-bezier(0.77, 0, 0.175, 1)' : 'none';
        } else if (index === prevIndex) {
          zIndex = 1; // Behind current
          transform = 'translateY(0%)'; // Stay put
        } else {
           zIndex = 0; // Hidden
        }
        
        // Initial Load State
        if (!isAnimating && index === currentIndex) {
            transform = 'translateY(0%)';
        }

        return (
          <div
            key={index}
            style={{
              position: 'absolute',
              top: 0,
              left: 0,
              width: '100%',
              height: '100%',
              zIndex: zIndex,
              transform: transform,
              transition: transition,
              willChange: 'transform'
            }}
          >
            <div style={{
              width: '100%',
              height: '100%',
              backgroundImage: `url(${slide.image})`,
              backgroundSize: 'cover',
              backgroundPosition: 'center',
              filter: 'grayscale(20%) contrast(1.1)' // Editorial touch
            }} />
          </div>
        );
      })}

      {/* --- TYPOGRAPHY LAYER (Fixed on top) --- */}
      <div style={{
        position: 'absolute',
        top: 0,
        left: 0,
        width: '100%',
        height: '100%',
        zIndex: 10,
        display: 'flex',
        flexDirection: 'column',
        justifyContent: 'center',
        alignItems: 'center',
        pointerEvents: 'none',
        mixBlendMode: 'difference', // THE MAGIC SAUCE
        color: '#FFFFFF'
      }}>
        {/* We animate the text content swap */}
        <div style={{ textAlign: 'center' }}>
          <p style={{
             fontSize: '14px',
             textTransform: 'uppercase',
             letterSpacing: '0.3em',
             marginBottom: '16px',
             opacity: isAnimating ? 0 : 1,
             transition: 'opacity 0.4s ease',
             fontFamily: 'Helvetica, Arial, sans-serif'
          }}>
            {slides[currentIndex].subtitle}
          </p>
          <h2 style={{
             fontSize: 'clamp(40px, 8vw, 120px)',
             fontWeight: '400',
             fontFamily: '"Times New Roman", Times, serif',
             margin: 0,
             lineHeight: 1,
             opacity: isAnimating ? 0 : 1,
             transform: isAnimating ? 'translateY(20px)' : 'translateY(0)',
             transition: 'opacity 0.5s ease 0.4s, transform 0.5s ease 0.4s'
          }}>
            {slides[currentIndex].title}
          </h2>
        </div>
      </div>

      {/* Progress / Indicators */}

        position: 'absolute',
        bottom: '40px',

        zIndex: 11,
        mixBlendMode: 'difference',
        color: '#fff',
        fontFamily: 'monospace',
        fontSize: '16px'
      }}>

      </div>

        position: 'absolute',
        bottom: '40px',

        transform: 'translateX(-50%)',

        textTransform: 'uppercase',
