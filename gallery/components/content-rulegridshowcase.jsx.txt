                      <span style={s.outputLabelHeader}>Final Values</span>
                    </div>
                    <div style={s.outputRow}>
                      <div style={s.outputItem}>
                        <div style={s.outputLabel}>Width</div>
                        <div style={s.outputValue}>{outputWidth}</div>
                      </div>
                      <div style={s.outputItem}>
                        <div style={s.outputLabel}>Height</div>
                        <div style={s.outputValue}>{outputHeight}</div>
                      </div>
                    </div>
                  </div>
                </div>
              )}
            </div>
          </div>
        );
        
      default:
        return (
          <div style={s.card}>
            <div style={s.header}>
              <span style={s.idBadge}>{ruleId}</span>
              <span style={s.status}>{ruleStatus}</span>
            </div>
            <div style={s.title}>{ruleElement}</div>
            <div style={s.subtitle}>{ruleCategory}</div>
          </div>
        );
    }
  };
  
  // ═══════════════════════════════════════════════════════════════
  // MAIN RENDER
  // ═══════════════════════════════════════════════════════════════
  
  return (
    <div 
      className="rule-grid-showcase"
      style={{
        fontFamily,
        fontSize,
        fontWeight,
        color: textPrimary,
        lineHeight: 1.5,
        background: backgroundColor,
        padding: padding,
        borderRadius: borderRadius,
      }}
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
    >
      {renderCard()}
    </div>
  );
}
