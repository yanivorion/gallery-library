  "description": "Mouse-reactive mood board with multi-plane depth parallax.",

    "data": {
      "images": {
        "dataType": "text",
        "displayName": "Image URLs (Comma Sep)",
        "defaultValue": "assets/05.webp,assets/06.webp,assets/07.webp,assets/08.webp,assets/01.webp",
        "group": "Content"
      },
      "intensity": {
        "dataType": "number",
        "displayName": "Movement Intensity",

        "dataType": "color",

      "resizeDirection": "horizontalAndVertical",

function Component({ config = {} }) {
  const containerRef = React.useRef(null);
  const [mousePos, setMousePos] = React.useState({ x: 0, y: 0 });
  const [dimensions, setDimensions] = React.useState({ width: 0, height: 0 });

  // Safe Config
  const rawImages = config?.images || '';
  const imageList = React.useMemo(() => rawImages.split(',').map(i => i.trim()).filter(Boolean), [rawImages]);
  const intensity = parseFloat(config?.intensity || '30');
  const bgColor = config?.backgroundColor || '#eeeeee';

  // Seeded Random Helper (Deterministic)
  const getSeededRandom = (seed) => {
    const x = Math.sin(seed++) * 10000;
    return x - Math.floor(x);
  };

  // Generate Layout Data Once
  const items = React.useMemo(() => {
    return imageList.map((url, i) => {
      const rng = (offset) => getSeededRandom(i * 10 + offset);
      
      // Depth: 1 (Close/Fast) to 0.2 (Far/Slow)
      const depth = 0.2 + (rng(1) * 0.8);
      
      return {
        id: i,
        url,
        depth,
        // Position percentage (10% to 90%)
        top: 10 + (rng(2) * 70), 
        left: 10 + (rng(3) * 70),
        // Size (150px to 350px)
        width: 150 + (rng(4) * 200),
        // Aspect Ratio (0.8 to 1.5)
        aspect: 0.8 + (rng(5) * 0.7),
        zIndex: Math.floor(depth * 10)
      };
    });
  }, [imageList]);

  // Mouse Tracking
  React.useEffect(() => {
    const container = containerRef.current;
    if (!container) return;

    // Set initial dimensions
    const updateDimensions = () => {
      setDimensions({
        width: container.offsetWidth,
        height: container.offsetHeight
      });
    };
    updateDimensions();
    window.addEventListener('resize', updateDimensions);

    const handleMouseMove = (e) => {
      const rect = container.getBoundingClientRect();
      const x = (e.clientX - rect.left) / rect.width - 0.5; // -0.5 to 0.5
      const y = (e.clientY - rect.top) / rect.height - 0.5;
      
      // Smooth interpolation could be added here, using raw values for responsiveness
      setMousePos({ x, y });
    };

    container.addEventListener('mousemove', handleMouseMove);
    return () => {
      window.removeEventListener('resize', updateDimensions);
      container.removeEventListener('mousemove', handleMouseMove);
    };
  }, []);

  const prefersReducedMotion = typeof window !== 'undefined' 
    ? window.matchMedia('(prefers-reduced-motion: reduce)').matches 
    : false;

  return (
    <div
      ref={containerRef}
      className="parallax-scatter"
      style={{
        width: '100%',
        height: '100%',
        minHeight: '600px',
        backgroundColor: bgColor,
        position: 'relative',
        overflow: 'hidden',
        cursor: 'default'
      }}
    >
      {/* Introduction Text (Background Layer) */}
      <div style={{
        position: 'absolute',
        top: '50%',
        left: '50%',
        transform: `translate(-50%, -50%) translate(${mousePos.x * -10}px, ${mousePos.y * -10}px)`,
        color: '#cccccc',
        fontSize: '12vw',
        fontWeight: '800',
        lineHeight: 1,
        pointerEvents: 'none',
        zIndex: 0,
        whiteSpace: 'nowrap',
        opacity: 0.5
      }}>
        GALLERY
      </div>

      {/* Floating Images */}
      {items.map((item) => {
        // Parallax calculation
        const moveX = prefersReducedMotion ? 0 : mousePos.x * item.depth * -intensity * 4;
        const moveY = prefersReducedMotion ? 0 : mousePos.y * item.depth * -intensity * 4;

        return (
          <div
            key={item.id}
            style={{
              position: 'absolute',
              top: `${item.top}%`,
              left: `${item.left}%`,
              width: `${item.width}px`,
              height: `${item.width * item.aspect}px`,
              zIndex: item.zIndex,
              transform: `translate(-50%, -50%) translate(${moveX}px, ${moveY}px)`,
              transition: 'transform 100ms ease-out', // Small lag for smoothness
              willChange: 'transform'
            }}
          >
            <div style={{
              width: '100%',
              height: '100%',
              backgroundImage: `url(${item.url})`,
              backgroundSize: 'cover',
              backgroundPosition: 'center',
              boxShadow: `0 10px 30px -10px rgba(0,0,0,${0.1 + (item.depth * 0.2)})`, // Deeper items = lighter shadow
              filter: `blur(${prefersReducedMotion ? 0 : (1 - item.depth) * 2}px)`, // Blur distant items
              opacity: 0,
              animation: `fadeIn 800ms ease-out ${item.id * 100}ms forwards`
            }} />
          </div>
        );
      })}
      
      <style>{`
        @keyframes fadeIn { from { opacity: 0; transform: scale(0.9); } to { opacity: 1; transform: scale(1); } }
      `}</style>
    </div>
  );
}
