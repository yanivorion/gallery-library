  const [isAnimating, setIsAnimating] = React.useState(false);
  const containerRef = React.useRef(null);
  
  const imagesRaw = config?.images || '';
  const parallaxStrength = config?.parallaxStrength || 'medium';
  const autoPlay = config?.autoPlay !== false;
  const interval = parseInt(config?.interval || '5000');
  const containerHeight = config?.containerHeight || '600';
  const showDepthLayers = config?.showDepthLayers !== false;
  const backgroundColor = config?.backgroundColor || '#0a0a1a';
  const overlayColor = config?.overlayColor || '#1a1a2e';
  const navColor = config?.navColor || '#ffffff';
  
  const images = React.useMemo(() => {
    return imagesRaw.split(',').map(url => url.trim()).filter(Boolean);
  }, [imagesRaw]);
  
  const prefersReducedMotion = React.useMemo(() => {
    return typeof window !== 'undefined' 
      ? window.matchMedia('(prefers-reduced-motion: reduce)').matches 
      : false;
  }, []);
  
  const strengthConfig = React.useMemo(() => {
    const configs = {
      subtle: { layer1: 10, layer2: 20, layer3: 30 },
      medium: { layer1: 20, layer2: 40, layer3: 60 },
      strong: { layer1: 30, layer2: 60, layer3: 90 }
    };
    return configs[parallaxStrength] || configs.medium;
  }, [parallaxStrength]);
  
  // Mouse tracking
  React.useEffect(() => {
    if (prefersReducedMotion) return;
    
    const container = containerRef.current;
    if (!container) return;
    
    const handleMouseMove = (e) => {
      const rect = container.getBoundingClientRect();
      const x = (e.clientX - rect.left) / rect.width;
      const y = (e.clientY - rect.top) / rect.height;
      setMousePosition({ x, y });
    };
    
    const handleMouseLeave = () => {
      setMousePosition({ x: 0.5, y: 0.5 });
    };
    
    container.addEventListener('mousemove', handleMouseMove);
    container.addEventListener('mouseleave', handleMouseLeave);
    
    return () => {
      container.removeEventListener('mousemove', handleMouseMove);
      container.removeEventListener('mouseleave', handleMouseLeave);
    };
  }, [prefersReducedMotion]);
  
  const getParallaxTransform = (strength) => {
    if (prefersReducedMotion) return 'none';
    const x = (mousePosition.x - 0.5) * strength;

    return `translate(${x}px, ${y}px)`;
  };
  
  const handleTransition = React.useCallback((nextIndex) => {
    if (isAnimating || images.length <= 1) return;
    
    setIsAnimating(true);

      setCurrentIndex(nextIndex);
      setIsAnimating(false);
    }, 600);
  }, [isAnimating, images.length]);

    if (!autoPlay || images.length <= 1 || prefersReducedMotion) return;
    
    const timer = setInterval(() => {
      handleTransition((currentIndex + 1) % images.length);
    }, interval);
    
    return () => clearInterval(timer);
  }, [autoPlay, interval, currentIndex, images.length, handleTransition, prefersReducedMotion]);

    const next = direction === 'next'

      : (currentIndex - 1 + images.length) % images.length;

            transition: 'opacity 600ms ease'
          }}
        />
      </div>
      
      {/* Middle layer */}
      {showDepthLayers && (
        <div
          style={{
            position: 'absolute',
            inset: '-40px',

            opacity: 0.5
          }}
        >
          <div
            style={{

              transition: 'background 0.3s ease-out'
            }}
          />

      )}

              }}

        }}
