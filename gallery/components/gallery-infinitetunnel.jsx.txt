function Component({ config = {} }) {
  const containerRef = React.useRef(null);
  const [scrollZ, setScrollZ] = React.useState(0);
  const targetZ = React.useRef(0);
  const currentZ = React.useRef(0);
  const animationFrame = React.useRef();

  // Safe Config
  const rawImages = config?.images || '';
  const images = React.useMemo(() => rawImages.split(',').map(i => i.trim()).filter(Boolean), [rawImages]);
  const densityMap = { sparse: 1000, medium: 600, dense: 400 };
  const zStep = densityMap[config?.tunnelLength || 'medium'];
  const speedMult = parseFloat(config?.scrollSpeed || '2');
  const fogColor = config?.fogColor || '#000000';

  // Generate 3D Positions
  const items = React.useMemo(() => {
    // Create enough items to fill a long tunnel, repeating images
    const totalItems = 30; 
    return Array.from({ length: totalItems }).map((_, i) => {
      const imgIndex = i % images.length;
      // Spiral distribution
      const angle = i * 0.5; 
      const radius = 350; // Distance from center
      return {
        id: i,
        url: images[imgIndex] || 'https://via.placeholder.com/400',
        x: Math.cos(angle) * radius,
        y: Math.sin(angle) * radius,
        z: -(i * zStep), // Stacking backwards
        rotation: (Math.random() * 20) - 10
      };
    });
  }, [images, zStep]);

  // Animation Loop for Smooth Scroll
  React.useEffect(() => {
    const loop = () => {
      // Linear interpolation for smooth momentum
      currentZ.current += (targetZ.current - currentZ.current) * 0.08;
      
      // Infinite loop logic: reset when we pass a certain threshold
      const tunnelDepth = items.length * zStep;
      if (currentZ.current > zStep) {
         // Seamless looping would require shifting array, 
         // for this demo we limit forward travel or let it fly through
      }
      
      setScrollZ(currentZ.current);
      animationFrame.current = requestAnimationFrame(loop);
    };
    loop();
    return () => cancelAnimationFrame(animationFrame.current);
  }, [items.length, zStep]);

  // Wheel Handler
  React.useEffect(() => {
    const container = containerRef.current;
    if (!container) return;

    const handleWheel = (e) => {
      e.preventDefault();
      // Forward (Scroll Down) = Increase Z (Camera moves into screen)
      targetZ.current += e.deltaY * speedMult;
      
      // Clamp backward movement to start
      if (targetZ.current < 0) targetZ.current = 0;
    };

    container.addEventListener('wheel', handleWheel, { passive: false });
    return () => container.removeEventListener('wheel', handleWheel);
  }, [speedMult]);

  // Accessibility: Reduced Motion
  const prefersReducedMotion = typeof window !== 'undefined' 
    ? window.matchMedia('(prefers-reduced-motion: reduce)').matches 
    : false;

  return (
    <div
      ref={containerRef}
      className="tunnel-gallery"
      style={{
        width: '100%',
        height: '100%',
        minHeight: '600px',
        backgroundColor: fogColor,
        overflow: 'hidden',
        position: 'relative',
        cursor: 'ns-resize', // Indicate scroll vertical
        perspective: '800px', // Critical for 3D
        perspectiveOrigin: '50% 50%'
      }}
    >
      {/* Introduction Overlay */}
      <div style={{
        position: 'absolute',
        bottom: '40px',
        left: '50%',
        transform: 'translateX(-50%)',
        color: '#FFFFFF',
        opacity: scrollZ > 200 ? 0 : 1,
        transition: 'opacity 500ms ease',
        textAlign: 'center',
        zIndex: 10,
        pointerEvents: 'none',
        fontFamily: 'sans-serif',
        textShadow: '0 2px 10px rgba(0,0,0,0.5)'
      }}>
        <p style={{ fontSize: '14px', letterSpacing: '2px', textTransform: 'uppercase' }}>Scroll to Explore</p>
        <div style={{ fontSize: '24px', marginTop: '8px' }}>↓</div>
      </div>

      {/* 3D World Container */}
      <div
        style={{
          position: 'absolute',
          top: '50%',
          left: '50%',
          width: '0px',
          height: '0px',
          transformStyle: 'preserve-3d',
          transform: `translate3d(0, 0, ${scrollZ}px)`
        }}
      >
        {items.map((item) => {
          // Calculate relative Z to camera
          const relativeZ = item.z + scrollZ;
          
          // Performance culling: Don't render if behind camera or too far
          const isVisible = relativeZ < 500 && relativeZ > -3000;
          if (!isVisible) return null;

          // Fog calculation
          const opacity = Math.min(1, (relativeZ + 2500) / 1000);

          return (
            <div
              key={item.id}
              style={{
                position: 'absolute',
                left: '-150px', // Center the 300px wide image
                top: '-200px', // Center the 400px high image
                width: '300px',
                height: '400px',
                transform: `
                  translate3d(${item.x}px, ${item.y}px, ${item.z}px) 
                  rotateZ(${prefersReducedMotion ? 0 : item.rotation}deg)
                `,
                opacity: opacity,
                transition: 'opacity 200ms linear',
                willChange: 'transform'
              }}
            >
              <div style={{
                width: '100%',
                height: '100%',
                backgroundColor: '#212529',
                backgroundImage: `url(${item.url})`,
                backgroundSize: 'cover',
                backgroundPosition: 'center',
                boxShadow: `0 0 30px ${fogColor}`, // Blend edges into fog
                borderRadius: '4px'
              }} />
            </div>
          );
        })}
      </div>
    </div>
  );
}
