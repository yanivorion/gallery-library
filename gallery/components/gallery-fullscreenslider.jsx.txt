    { url: "assets/05.webp", caption: "Alpine Lake" }
  ]);
  
  const images = typeof imagesData === 'string' ? JSON.parse(imagesData) : imagesData;
  const backgroundColor = config?.backgroundColor || "#111111";
  const thumbnailStripBackground = config?.thumbnailStripBackground || "#222222";
  const controlsColor = config?.controlsColor || "#ffffff";
  const accentColor = config?.accentColor || "#777777";
  const thumbnailBorderColor = config?.thumbnailBorderColor || "#777777";
  const showCaptions = config?.showCaptions !== false;
  const showCounter = config?.showCounter !== false;
  const thumbnailSize = parseInt(config?.thumbnailSize || "80");
  const thumbnailGap = parseInt(config?.thumbnailGap || "12");
  const autoPlay = config?.autoPlay === true;
  const autoPlayInterval = parseInt(config?.autoPlayInterval || "4000");
  const transitionDuration = parseInt(config?.transitionDuration || "400");

  const [currentIndex, setCurrentIndex] = React.useState(0);
  const [isTransitioning, setIsTransitioning] = React.useState(false);
  const thumbnailStripRef = React.useRef(null);

  const prefersReducedMotion = typeof window !== 'undefined'
    ? window.matchMedia('(prefers-reduced-motion: reduce)').matches
    : false;

  const duration = prefersReducedMotion ? 0 : transitionDuration;

  // Navigation functions
  const goToNext = React.useCallback(() => {
    if (isTransitioning) return;
    setIsTransitioning(true);
    setCurrentIndex((prev) => (prev + 1) % images.length);
    setTimeout(() => setIsTransitioning(false), duration);
  }, [isTransitioning, images.length, duration]);

  const goToPrevious = React.useCallback(() => {
    if (isTransitioning) return;
    setIsTransitioning(true);
    setCurrentIndex((prev) => (prev - 1 + images.length) % images.length);
    setTimeout(() => setIsTransitioning(false), duration);
  }, [isTransitioning, images.length, duration]);

  const goToIndex = (index) => {
    if (isTransitioning || index === currentIndex) return;
    setIsTransitioning(true);
    setCurrentIndex(index);
    setTimeout(() => setIsTransitioning(false), duration);
  };

  // Keyboard navigation
  React.useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === 'ArrowLeft') {
        e.preventDefault();
        goToPrevious();
      } else if (e.key === 'ArrowRight') {
        e.preventDefault();
        goToNext();
      }
    };

    if (typeof window !== 'undefined') {
      window.addEventListener('keydown', handleKeyDown);
      return () => window.removeEventListener('keydown', handleKeyDown);
    }
  }, [goToNext, goToPrevious]);

  // Auto play
  React.useEffect(() => {
    if (!autoPlay) return;

    const interval = setInterval(goToNext, autoPlayInterval);
    return () => clearInterval(interval);
  }, [autoPlay, autoPlayInterval, goToNext]);

    if (thumbnailStripRef.current) {

        activeThumbnail.scrollIntoView({
          behavior: prefersReducedMotion ? 'auto' : 'smooth',

          inline: 'center'

            e.currentTarget.style.transform = 'translateY(-50%) scale(1)';

        >

        )}

        {showCaptions && currentImage.caption && (
