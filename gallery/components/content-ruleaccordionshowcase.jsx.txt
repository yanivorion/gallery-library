                <div style={s.outputGrid}>
                  <div style={s.outputItem}>
                    <div style={s.outputItemLabel}>Width</div>
                    <div style={s.outputValue}>{outputWidth}</div>
                  </div>
                  <div style={s.outputItem}>
                    <div style={s.outputItemLabel}>Height</div>
                    <div style={s.outputValue}>{outputHeight}</div>
                  </div>
                </div>
              </div>
            </div>
          </div>
        );
        
      default: // modern-cards, minimal-lines
        return (
          <div style={s.container}>
            {s.meta && (
              <div style={s.meta}>
                {['ID', 'Category', 'Element', 'Status'].map((label, idx) => (
                  <div key={idx} style={s.metaItem || {}}>
                    <span style={s.metaLabel}>{label}</span>
                    <span style={s.metaValue}>
                      {label === 'ID' ? ruleId : 
                       label === 'Category' ? ruleCategory :
                       label === 'Element' ? ruleElement : ruleStatus}
                    </span>
                  </div>
                ))}
              </div>
            )}
            <div style={s.stepsContainer}>
              {steps.map((step, idx) => (
                <div key={idx} style={step.isOutput ? s.outputCard : s.stepCard}>
                  {showStepNumbers && (
                    preset === 'minimal-lines' ? (
                      <span style={s.stepNumber}>[{step.num}]</span>
                    ) : (
                      <div style={s.stepNumber}>{step.num}</div>
                    )
                  )}
                  <div style={s.stepContent}>
                    <div style={s.stepLabel}>{step.label}</div>
                    {step.isOutput ? (
                      <div style={s.outputGrid}>
                        <div style={s.outputItem}>
                          <div style={s.outputLabel}>Width</div>
                          <div style={s.outputValue}>{step.width}</div>
                        </div>
                        <div style={s.outputItem}>
                          <div style={s.outputLabel}>Height</div>
                          <div style={s.outputValue}>{step.height}</div>
                        </div>
                      </div>
                    ) : (
                      <div style={s.stepValue}>{step.value}</div>
                    )}
                  </div>
                </div>
              ))}
            </div>
          </div>
        );
    }
  };
  
  // ═══════════════════════════════════════════════════════════════
  // MAIN RENDER
  // ═══════════════════════════════════════════════════════════════
  
  return (
    <div 
      className="rule-accordion-showcase"
      style={{
        fontFamily,
        fontSize,
        fontWeight,
        color: textPrimary,
        lineHeight: 1.5,
        ...styles.container,
      }}
    >
      {/* Outer Row (Collapsed State) */}
      <div
        onClick={() => setIsExpanded(!isExpanded)}
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => setIsHovered(false)}
        role="button"
        tabIndex={0}
        aria-expanded={isExpanded}
        aria-label={`Rule ${ruleId}: ${ruleElement}. Click to ${isExpanded ? 'collapse' : 'expand'}`}
        onKeyDown={(e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            setIsExpanded(!isExpanded);
          }
        }}
      >
        {renderOuterRow()}
      </div>
      
      {/* Inner Content (Expanded State) - CSS Grid Animation */}
      <div
        style={{
          display: 'grid',
          gridTemplateRows: isExpanded ? '1fr' : '0fr',
          transition: `grid-template-rows ${duration}ms cubic-bezier(0.4, 0, 0.2, 1)`,
        }}
      >
        <div style={{ overflow: 'hidden' }}>
          {renderInnerContent()}
        </div>
      </div>
    </div>
  );
}
