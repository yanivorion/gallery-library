function Component({ config = {} }) {
  const cardsJSON = config?.cards || "[]";
  const characterIcon = config?.characterIcon || "🎨";
  const showCharacterIcon = config?.showCharacterIcon !== false;
  const stageWidth = parseInt(config?.stageWidth || "400");
  const stageHeight = parseInt(config?.stageHeight || "500");
  const backgroundColor = config?.backgroundColor || "#eeeeee";
  const uiBackgroundColor = config?.uiBackgroundColor || "#dddddd";
  const uiBorderColor = config?.uiBorderColor || "#aaaaaa";
  const stageBorderColor = config?.stageBorderColor || "#bbbbbb";
  const textColor = config?.textColor || "#222222";
  const transitionDuration = parseInt(config?.transitionDuration || "400");
  const autoPlay = config?.autoPlay === true;
  const autoPlayInterval = parseInt(config?.autoPlayInterval || "3") * 1000;

  const [cards, setCards] = React.useState([]);
  const [activeCardIndex, setActiveCardIndex] = React.useState(0);
  const [isMobile, setIsMobile] = React.useState(false);

  // Parse cards
  React.useEffect(() => {
    try {
      const parsed = JSON.parse(cardsJSON);
      setCards(Array.isArray(parsed) ? parsed : []);
    } catch (e) {
      setCards([]);
    }
  }, [cardsJSON]);

  // Check mobile
  React.useEffect(() => {
    if (typeof window !== 'undefined') {
      setIsMobile(window.innerWidth < 768);
      const handleResize = () => setIsMobile(window.innerWidth < 768);
      window.addEventListener('resize', handleResize);
      return () => window.removeEventListener('resize', handleResize);
    }
  }, []);

  // Auto-play
  React.useEffect(() => {
    if (!autoPlay || cards.length === 0) return;

    const interval = setInterval(() => {
      setActiveCardIndex((prev) => (prev + 1) % cards.length);
    }, autoPlayInterval);

    return () => clearInterval(interval);
  }, [autoPlay, cards.length, autoPlayInterval]);

  const handleNext = () => {
    setActiveCardIndex((prev) => (prev + 1) % cards.length);
  };

  const handlePrev = () => {
    setActiveCardIndex((prev) => (prev - 1 + cards.length) % cards.length);
  };

  const currentStageWidth = isMobile ? stageWidth * 0.75 : stageWidth;
  const currentStageHeight = isMobile ? stageHeight * 0.8 : stageHeight;

  return (
    <div
      className="card-showcase-simple"
      style={{
        position: 'relative',
        minHeight: '100vh',
        backgroundColor: backgroundColor,
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        fontFamily: "'Neue Haas Grotesk Display Pro', 'Neue Haas Grotesk Text Pro', 'Helvetica Neue', Helvetica, Arial, sans-serif"
      }}
    >
      {/* Character Icon */}
      {showCharacterIcon && (
        <div
          style={{
            position: 'fixed',
            top: isMobile ? '16px' : '20px',
            left: '20px',
            width: '40px',
            height: '40px',
            backgroundColor: uiBackgroundColor,
            border: `1px solid ${uiBorderColor}`,
            borderRadius: '4px',
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            fontSize: '24px',
            zIndex: 100
          }}
        >
          {characterIcon}
        </div>
      )}

      {/* Card Counter */}
      <div
        style={{
          position: 'fixed',
          top: isMobile ? '16px' : '20px',
          right: '20px',
          backgroundColor: uiBackgroundColor,
          border: `1px solid ${uiBorderColor}`,
          borderRadius: '4px',
          padding: '8px 16px',
          fontSize: '14px',
          fontWeight: '600',
          color: textColor,
          zIndex: 100
        }}
      >
        {activeCardIndex + 1} / {cards.length}
      </div>

      {/* Info Card */}
      {cards.length > 0 && (
        <div
          style={{
            position: 'fixed',
            left: '50%',
            bottom: '30px',
            transform: 'translateX(-50%)',
            display: 'flex',
            alignItems: 'center',
            gap: '10px',
            width: isMobile ? '260px' : '300px',
            backgroundColor: uiBackgroundColor,
            border: `1px solid ${uiBorderColor}`,
            borderRadius: '4px',
            padding: '3px',
            zIndex: 100
          }}
        >
          <div
            style={{
              width: '40px',
              height: '40px',
              borderRadius: '4px',
              border: `1px solid ${uiBorderColor}`,
              overflow: 'hidden',
              flexShrink: 0
            }}
          >
            <img
              src={cards[activeCardIndex]?.image || ''}
              alt=""
              style={{
                width: '110%',
                height: '110%',
                objectFit: 'cover'
              }}
            />
          </div>
          <div
            style={{
              fontSize: isMobile ? '13px' : '14px',
              fontWeight: '600',
              color: textColor,
              overflow: 'hidden',
              textOverflow: 'ellipsis',
              whiteSpace: 'nowrap'
            }}
          >
            {cards[activeCardIndex]?.title || 'Untitled'}
          </div>
        </div>
      )}

      {/* Stage */}
      <div
        style={{
          position: 'relative',
          width: `${currentStageWidth}px`,
          height: `${currentStageHeight}px`,
          border: `1px solid ${stageBorderColor}`
        }}
      >
        {/* Cards */}
        {cards.map((card, index) => (
          <div
            key={index}
            style={{
              position: 'absolute',
              top: '50%',
              left: '50%',
              transform: 'translate(-50%, -50%)',
              width: '95%',
              height: '95%',
              display: 'flex',
              alignItems: 'center',
              justifyContent: 'center',
              overflow: 'hidden',
              clipPath: index <= activeCardIndex ? 'inset(0%)' : 'inset(100%)',
              transition: `clip-path ${transitionDuration}ms ease`
            }}
          >
            <img
              src={card.image}
              alt={card.title}
              style={{
                width: '100%',
                height: '100%',
                objectFit: 'cover'
              }}
            />
          </div>
        ))}

        {/* Navigation Arrows */}
        <button
          onClick={handlePrev}
          style={{
            position: 'absolute',
            left: '-60px',
            top: '50%',
            transform: 'translateY(-50%)',
            width: '44px',
            height: '44px',
            backgroundColor: uiBackgroundColor,
            border: `1px solid ${uiBorderColor}`,
            borderRadius: '4px',
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            cursor: 'pointer',
            transition: `all ${transitionDuration}ms ease`,
            fontSize: '20px'
          }}
          onMouseEnter={(e) => {
            e.currentTarget.style.backgroundColor = '#ffffff';
          }}
          onMouseLeave={(e) => {
            e.currentTarget.style.backgroundColor = uiBackgroundColor;
          }}
        >
          ←
        </button>

        <button
          onClick={handleNext}
          style={{
            position: 'absolute',
            right: '-60px',
            top: '50%',
            transform: 'translateY(-50%)',
            width: '44px',
            height: '44px',
            backgroundColor: uiBackgroundColor,
            border: `1px solid ${uiBorderColor}`,
            borderRadius: '4px',
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            cursor: 'pointer',
            transition: `all ${transitionDuration}ms ease`,
            fontSize: '20px'
          }}
          onMouseEnter={(e) => {
            e.currentTarget.style.backgroundColor = '#ffffff';
          }}
          onMouseLeave={(e) => {
            e.currentTarget.style.backgroundColor = uiBackgroundColor;
          }}
        >
          →
        </button>
      </div>

      {/* Keyboard Hint */}
      <div
        style={{
          position: 'fixed',
          bottom: '100px',
          left: '50%',
          transform: 'translateX(-50%)',
          fontSize: '11px',
          color: textColor,
          opacity: 0.4,
          textTransform: 'uppercase',
          letterSpacing: '0.1em'
        }}
      >
        Use arrows or click
      </div>
    </div>
  );
}
