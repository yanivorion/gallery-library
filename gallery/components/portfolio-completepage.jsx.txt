function Component({ config = {} }) {
  // Parse all config values
  const heroTitleEn = config?.heroTitleEn || "Exploration";
  const heroTitleJa = config?.heroTitleJa || "探求";
  const heroTextEn = config?.heroTextEn || "20 experimental works featuring test code and strong elements of research.";
  const heroTextJa = config?.heroTextJa || "試験的なコードと研究要素を備えた20作品";
  const videoUrl = config?.videoUrl || "";
  const projectsJSON = config?.projects || "[]";
  const nextSectionTitle = config?.nextSectionTitle || "Growth";
  const nextSectionLink = config?.nextSectionLink || "/growth/";
  const nextSectionImage = config?.nextSectionImage || "";
  const backgroundColor = config?.backgroundColor || "#0A0A0A";
  const accentColor = config?.accentColor || "#3B82F6";
  const webglColor = config?.webglColor || "#3B82F6";
  const javascriptColor = config?.javascriptColor || "#10B981";
  const cssColor = config?.cssColor || "#8B5CF6";
  const showJapanese = config?.showJapanese !== false;
  const enableFiltering = config?.enableFiltering !== false;
  const gridColumns = parseInt(config?.gridColumns || "4");

  const [projects, setProjects] = React.useState([]);
  const [selectedCategory, setSelectedCategory] = React.useState("All");
  const [visibleCards, setVisibleCards] = React.useState([]);
  const [heroVisible, setHeroVisible] = React.useState(false);
  const [videoLoaded, setVideoLoaded] = React.useState(false);
  const [isMobile, setIsMobile] = React.useState(false);
  
  const videoRef = React.useRef(null);
  const gridRef = React.useRef(null);

  // Parse projects JSON
  React.useEffect(() => {
    try {
      const parsed = JSON.parse(projectsJSON);
      setProjects(Array.isArray(parsed) ? parsed : []);
    } catch (e) {
      console.error('Invalid projects JSON');
      setProjects([]);
    }
  }, [projectsJSON]);

  // Check mobile
  React.useEffect(() => {
    if (typeof window !== 'undefined') {
      setIsMobile(window.innerWidth < 768);
      const handleResize = () => setIsMobile(window.innerWidth < 768);
      window.addEventListener('resize', handleResize);
      return () => window.removeEventListener('resize', handleResize);
    }
  }, []);

  // Hero entrance animation
  React.useEffect(() => {
    setTimeout(() => setHeroVisible(true), 100);
  }, []);

  // Grid intersection observer
  React.useEffect(() => {
    if (!gridRef.current) return;

    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            projects.forEach((_, index) => {
              setTimeout(() => {
                setVisibleCards((prev) => [...new Set([...prev, index])]);
              }, index * 50);
            });
            observer.disconnect();
          }
        });
      },
      { threshold: 0.1 }
    );

    observer.observe(gridRef.current);
    return () => observer.disconnect();
  }, [projects]);

  const categories = ["All", ...new Set(projects.map(p => p.category).filter(Boolean))];
  const filteredProjects = selectedCategory === "All"
    ? projects
    : projects.filter(p => p.category === selectedCategory);

  const getCategoryColor = (category) => {
    switch (category?.toLowerCase()) {
      case 'webgl': return webglColor;
      case 'javascript': return javascriptColor;
      case 'css': return cssColor;
      default: return '#737373';
    }
  };

  const getAnimationStyle = (delay = 0, isVisible = true) => ({
    opacity: isVisible ? 1 : 0,
    transform: isVisible ? 'translateY(0)' : 'translateY(30px)',
    transition: `opacity 600ms cubic-bezier(0.4, 0, 0.2, 1) ${delay}ms, transform 600ms cubic-bezier(0.4, 0, 0.2, 1) ${delay}ms`
  });

  return (
    <div
      className="portfolio-complete-page"
      style={{
        backgroundColor: backgroundColor,
        fontFamily: '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif'
      }}
    >
      {/* HERO SECTION */}
      <section
        style={{
          position: 'relative',
          minHeight: '100vh',
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          overflow: 'hidden'
        }}
      >
        {/* Video Background */}
        {videoUrl && !isMobile && (
          <video
            ref={videoRef}
            autoPlay
            loop
            muted
            playsInline
            onLoadedData={() => setVideoLoaded(true)}
            style={{
              position: 'absolute',
              top: '50%',
              left: '50%',
              minWidth: '100%',
              minHeight: '100%',
              width: 'auto',
              height: 'auto',
              transform: 'translate(-50%, -50%)',
              objectFit: 'cover',
              opacity: videoLoaded ? 1 : 0,
              transition: 'opacity 1s ease-out'
            }}
          >
            <source src={videoUrl} type="video/mp4" />
          </video>
        )}

        {/* Overlay */}
        <div
          style={{
            position: 'absolute',
            top: 0,
            left: 0,
            right: 0,
            bottom: 0,
            backgroundColor: '#000000',
            opacity: 0.6
          }}
        />

        {/* Hero Content */}
        <div style={{ position: 'relative', zIndex: 2, textAlign: 'center', padding: '80px 24px', maxWidth: '1200px' }}>
          <div style={{ ...getAnimationStyle(0, heroVisible), width: '60px', height: '3px', backgroundColor: accentColor, margin: '0 auto 32px' }} />
          
          <h1 style={{ ...getAnimationStyle(200, heroVisible), fontSize: 'clamp(40px, 8vw, 80px)', fontWeight: '300', color: '#FFFFFF', margin: '0 0 16px 0', letterSpacing: '0.02em', textTransform: 'uppercase' }}>
            {heroTitleEn}
          </h1>
          
          {showJapanese && (
            <h2 style={{ ...getAnimationStyle(400, heroVisible), fontSize: 'clamp(20px, 4vw, 40px)', fontWeight: '300', color: '#FFFFFF', margin: '0 0 40px 0', opacity: 0.8 }}>
              {heroTitleJa}
            </h2>
          )}
          
          <p style={{ ...getAnimationStyle(600, heroVisible), fontSize: 'clamp(16px, 2.5vw, 20px)', color: '#A3A3A3', lineHeight: '1.7', margin: '0 0 12px 0' }}>
            {heroTextEn}
          </p>
          
          {showJapanese && (
            <p style={{ ...getAnimationStyle(800, heroVisible), fontSize: 'clamp(14px, 2vw, 16px)', color: '#A3A3A3', lineHeight: '1.7', opacity: 0.7 }}>
              {heroTextJa}
            </p>
          )}
        </div>
      </section>

      {/* PROJECT GRID SECTION */}
      <section ref={gridRef} style={{ padding: 'clamp(48px, 8vw, 120px) clamp(24px, 5vw, 80px)' }}>
        <div style={{ maxWidth: '1600px', margin: '0 auto' }}>
          
          {/* Category Filter */}
          {enableFiltering && categories.length > 1 && (
            <div style={{ display: 'flex', gap: '12px', marginBottom: '48px', flexWrap: 'wrap', justifyContent: 'center' }}>
              {categories.map((category) => (
                <button
                  key={category}
                  onClick={() => setSelectedCategory(category)}
                  style={{
                    padding: '10px 24px',
                    backgroundColor: selectedCategory === category ? getCategoryColor(category) : 'transparent',
                    color: selectedCategory === category ? '#FFFFFF' : '#A3A3A3',
                    border: `1px solid ${selectedCategory === category ? getCategoryColor(category) : '#404040'}`,
                    borderRadius: '12px',
                    fontSize: '14px',
                    fontWeight: '500',
                    cursor: 'pointer',
                    transition: 'all 300ms ease-out',
                    textTransform: 'uppercase',
                    letterSpacing: '0.05em',
                    fontFamily: 'inherit'
                  }}
                >
                  {category}
                </button>
              ))}
            </div>
          )}

          {/* Grid */}
          <div style={{ display: 'grid', gridTemplateColumns: `repeat(auto-fit, minmax(280px, 1fr))`, gap: '24px' }}>
            {filteredProjects.map((project, index) => (
              <div
                key={index}
                onClick={() => window.open(project.url, '_blank')}
                style={{
                  backgroundColor: '#1A1A1A',
                  borderRadius: '12px',
                  padding: '32px 24px',
                  cursor: 'pointer',
                  transition: 'all 300ms cubic-bezier(0.4, 0, 0.2, 1)',
                  boxShadow: '0 4px 12px rgba(0,0,0,0.3)',
                  opacity: visibleCards.includes(index) ? 1 : 0,
                  transform: visibleCards.includes(index) ? 'translateY(0)' : 'translateY(20px)'
                }}
                onMouseEnter={(e) => {
                  e.currentTarget.style.transform = 'scale(1.05)';
                  e.currentTarget.style.boxShadow = '0 8px 24px rgba(0,0,0,0.4)';
                }}
                onMouseLeave={(e) => {
                  e.currentTarget.style.transform = 'scale(1)';
                  e.currentTarget.style.boxShadow = '0 4px 12px rgba(0,0,0,0.3)';
                }}
              >
                {project.category && (
                  <div style={{ display: 'inline-block', padding: '4px 12px', backgroundColor: `${getCategoryColor(project.category)}20`, color: getCategoryColor(project.category), borderRadius: '6px', fontSize: '11px', fontWeight: '600', textTransform: 'uppercase', letterSpacing: '0.1em', marginBottom: '16px' }}>
                    {project.category}
                  </div>
                )}
                <h3 style={{ fontSize: 'clamp(18px, 3vw, 22px)', fontWeight: '500', color: '#FFFFFF', lineHeight: '1.3', margin: '0 0 8px 0' }}>
                  {project.title}
                </h3>
                <p style={{ fontSize: '14px', color: '#737373', margin: 0 }}>
                  {project.date}
                </p>
              </div>
            ))}
          </div>
        </div>
      </section>

      {/* NEXT SECTION */}
      <section style={{ padding: 'clamp(48px, 8vw, 120px) clamp(24px, 5vw, 80px)' }}>
        <div style={{ maxWidth: '1600px', margin: '0 auto' }}>
          <div
            onClick={() => window.location.href = nextSectionLink}
            style={{
              position: 'relative',
              height: '500px',
              backgroundColor: '#1A1A1A',
              borderRadius: '16px',
              overflow: 'hidden',
              cursor: 'pointer',
              transition: 'all 400ms cubic-bezier(0.4, 0, 0.2, 1)',
              backgroundImage: nextSectionImage ? `url(${nextSectionImage})` : 'none',
              backgroundSize: 'cover',
              backgroundPosition: 'center',
              display: 'flex',
              alignItems: 'center',
              justifyContent: 'center'
            }}
            onMouseEnter={(e) => {
              e.currentTarget.style.transform = 'scale(1.02)';
            }}
            onMouseLeave={(e) => {
              e.currentTarget.style.transform = 'scale(1)';
            }}
          >
            <div style={{ position: 'absolute', top: 0, left: 0, right: 0, bottom: 0, backgroundColor: '#000000', opacity: 0.5 }} />
            <div style={{ position: 'relative', zIndex: 1, textAlign: 'center' }}>
              <p style={{ fontSize: '18px', color: '#A3A3A3', textTransform: 'uppercase', letterSpacing: '0.15em', margin: '0 0 16px 0' }}>
                Explore Next Section
              </p>
              <h2 style={{ fontSize: 'clamp(32px, 6vw, 64px)', fontWeight: '600', color: '#FFFFFF', margin: 0 }}>
                {nextSectionTitle}
              </h2>
            </div>
          </div>
        </div>
      </section>
    </div>
  );
}
